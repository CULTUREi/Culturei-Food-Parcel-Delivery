const twilio = require('twilio');
const admin = require('firebase-admin');

// Initialize
const client = twilio(process.env.TWILIO_SID, process.env.TWILIO_AUTH_TOKEN);
const VERIFY_SID = 'VA...'; // Your Service SID

// API Endpoint
app.post('/send-otp', async (req, res) => {
  const { phone, role } = req.body; // role = "customer" or "driver"
  
  try {
    // Send OTP via Twilio Verify
    const verification = await client.verify.v2.services(VERIFY_SID)
      .verifications.create({
        to: phone,
        channel: 'sms'
      });
    
    // Store in Firestore
    await admin.firestore().collection('otps').add({
      phone,
      role,
      status: 'pending',
      createdAt: admin.firestore.FieldValue.serverTimestamp()
    });
    
    res.json({ success: true, message: 'OTP sent!' });
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

// Verify OTP endpoint
app.post('/verify-otp', async (req, res) => {
  const { phone, code } = req.body;
  
  try {
    const verificationCheck = await client.verify.v2.services(VERIFY_SID)
      .verificationChecks.create({
        to: phone,
        code: code
      });
    
    if (verificationCheck.status === 'approved') {
      // Update Firestore
      await admin.firestore().collection('otps')
        .where('phone', '==', phone)
        .where('status', '==', 'pending')
        .get()
        .then(snapshot => {
          snapshot.forEach(doc => doc.ref.update({ status: 'verified' }));
        });
      
      res.json({ success: true, role: 'customer' }); // or fetch role from DB
    } else {
      res.status(400).json({ error: 'Invalid code' });
    }
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});
